Ustawienia symulacji

    \begin{algorithm}
    \caption{FindMin}
    \begin{algorithmic}
    \FUNCTION{FindMin}{A, n}
        \STATE $min \gets A[0]$
        \FOR{$i \gets 1$ \TO $n - 1$}
            \IF{$A[i] < min$}
                \STATE $min \gets A[i]$
            \ENDIF
        \ENDFOR
        \RETURN $min$
    \ENDFUNCTION
    \end{algorithmic}
    \end{algorithm}
