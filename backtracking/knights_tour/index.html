Problem skoczka szachowego — przeszukiwanie z powrotami

Skoczek odwiedza każde pole dokładnie raz. Przy ślepym zaułku algorytm cofa ruch i próbuje innej drogi.

    \begin{algorithm}
    \caption{KnightsTour}
    \begin{algorithmic}
    \FUNCTION{KnightsTour}{square, visited}
        \IF{all squares are visited}
            \RETURN true
        \ENDIF
        \FOR{each legal unvisited knight move $next$}
            \STATE $visited[next] \gets true$
            \IF{KnightsTour(next, visited)}
                \RETURN true
            \ENDIF
            \STATE $visited[next] \gets false$
        \ENDFOR
        \RETURN false
    \ENDFUNCTION
    \end{algorithmic}
    \end{algorithm}
