    \begin{algorithm}
    \caption{LowerBound}
    \begin{algorithmic}
    \FUNCTION{LowerBound}{A, n, target}
        \STATE $left \gets 0$
        \STATE $right \gets n - 1$
        \STATE $result \gets n$
        \WHILE{$left \le right$}
            \STATE $middle \gets \lfloor \frac{left + right}{2} \rfloor$
            \IF{$A[middle] < target$}
                \STATE $left \gets middle + 1$
            \ELSE
                \STATE $result \gets middle$
                \STATE $right \gets middle - 1$
            \ENDIF
        \ENDWHILE
        \RETURN $result$
    \ENDFUNCTION
    \end{algorithmic}
    \end{algorithm}
